cmake_minimum_required(VERSION 3.16)
project(fileman LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)

set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_FLAGS " -O3 -flto -pipe -g -Wp,-D_FORTIFY_SOURCE=3 -Wp,-D_GLIBCXX_ASSERTIONS -fexceptions -fstack-protector-strong -grecord-gcc-switches -fasynchronous-unwind-tables -fstack-clash-protection -fcf-protection -mno-omit-leaf-frame-pointer -march=native ")
set(LDFLAGS " -Wl,-z,relro,-z,now -Wl,--as-needed ")
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)

qt_add_executable(fileman main.cpp fileman.cpp fileman.h)
target_link_libraries(fileman PRIVATE Qt6::Core)
target_compile_definitions(fileman PRIVATE QT_ENABLE_STRICT_MODE_UP_TO=0x060900)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
